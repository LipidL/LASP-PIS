cmake_minimum_required(VERSION 3.24)
project(chemio VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
add_library(chemio STATIC src/arc.cpp)
add_library(chemio::chemio ALIAS chemio)
target_compile_features(chemio PUBLIC cxx_std_17)
target_include_directories(chemio PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(chemio PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
install(TARGETS chemio EXPORT chemioTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT chemioTargets NAMESPACE chemio:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/chemio)
configure_package_config_file(cmake/chemioConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/chemioConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/chemio)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/chemioConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/chemioConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/chemioConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/chemio)
if(BUILD_TESTING)
  add_executable(chemio_tests tests/arc.cpp)
  target_link_libraries(chemio_tests PRIVATE chemio::chemio)
  add_test(NAME chemio.arc COMMAND chemio_tests)
endif()
