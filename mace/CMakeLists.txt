cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES "80;86;89;90" CACHE STRING "CUDA architectures")
endif()
project(periodic_pairwise VERSION 0.1.0 LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
find_package(CUDAToolkit REQUIRED)
add_library(pairwise SHARED src/runtime.cu src/lj.cu src/mh1.cu src/c_api.cpp)
target_include_directories(pairwise PUBLIC include)
target_link_libraries(pairwise PRIVATE CUDA::cudart CUDA::cublas)
target_compile_options(pairwise PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr;-lineinfo>)
set_target_properties(pairwise PROPERTIES POSITION_INDEPENDENT_CODE ON)
option(PAIRWISE_BUILD_EXPLICIT_REFERENCE "Build optional explicit-list benchmark library" OFF)
if(PAIRWISE_BUILD_EXPLICIT_REFERENCE)
  add_library(pairwise_explicit SHARED src/runtime.cu src/lj.cu src/mh1.cu src/c_api.cpp src/explicit_reference.cu)
  target_include_directories(pairwise_explicit PUBLIC include)
  target_link_libraries(pairwise_explicit PRIVATE CUDA::cudart CUDA::cublas)
  target_compile_definitions(pairwise_explicit PRIVATE PW_EXPLICIT_REFERENCE=1)
  target_compile_options(pairwise_explicit PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr;-lineinfo>)
endif()
add_executable(pairwise_tests tests/framework.cu)
target_link_libraries(pairwise_tests PRIVATE pairwise CUDA::cudart)
enable_testing()
add_test(NAME framework COMMAND pairwise_tests)
install(TARGETS pairwise LIBRARY DESTINATION lib)
install(DIRECTORY include/pairwise DESTINATION include)
