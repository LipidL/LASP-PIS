cmake_minimum_required(VERSION 3.24)
project(checkpoint VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
find_package(ZLIB REQUIRED)
add_library(checkpoint STATIC src/checkpoint.cpp)
add_library(checkpoint::checkpoint ALIAS checkpoint)
target_compile_features(checkpoint PUBLIC cxx_std_17)
target_include_directories(checkpoint PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(checkpoint PUBLIC ZLIB::ZLIB)
set_target_properties(checkpoint PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
install(TARGETS checkpoint EXPORT checkpointTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT checkpointTargets NAMESPACE checkpoint:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/checkpoint)
configure_package_config_file(cmake/checkpointConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/checkpointConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/checkpoint)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/checkpointConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/checkpointConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/checkpointConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/checkpoint)
if(BUILD_TESTING)
  add_executable(checkpoint_tests tests/reader.cpp)
  target_link_libraries(checkpoint_tests PRIVATE checkpoint::checkpoint)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME checkpoint.reader COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reader.py $<TARGET_FILE:checkpoint_tests>)
  add_executable(checkpoint_inspect tests/inspect.cpp)
  target_link_libraries(checkpoint_inspect PRIVATE checkpoint::checkpoint)
endif()
